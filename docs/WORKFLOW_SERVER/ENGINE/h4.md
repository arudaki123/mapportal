# Example Import Module C++
    import module ngoài sử dụng c++ 
  VBD Workflow Engine sử dụng v8 engine để compile script js. 
```js
import { intersect } from "@turf/intersect";
import {polygon,featureCollection} from '@turf/helpers';
import { union } from "@turf/union";

var poly1 = polygon([
  [
    [-122.801742, 45.48565],
    [-122.801742, 45.60491],
    [-122.584762, 45.60491],
    [-122.584762, 45.48565],
    [-122.801742, 45.48565],
  ],
]);

var poly2 = polygon([
  [
    [-122.520217, 45.535693],
    [-122.64038, 45.553967],
    [-122.720031, 45.526554],
    [-122.669906, 45.507309],
    [-122.723464, 45.446643],
    [-122.532577, 45.408574],
    [-122.487258, 45.477466],
    [-122.520217, 45.535693],
  ],
]);

var result_union = union(featureCollection([poly1, poly2]));
console.log(JSON.stringify(result_union)); 
console.log(JSON.stringify(featureCollection(result_union))); 
var result = intersect(featureCollection([poly1, poly2]));
activity.setOutput(result); 
```
